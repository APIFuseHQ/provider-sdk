import type { OperationErrorCode } from "@apifuse/provider-sdk";
import { ProviderError, defineOperation } from "@apifuse/provider-sdk/provider";

import type { ProviderContext } from "../index";

const SESSION_ERROR_CODES = [
	{ code: "UPSTREAM_SCHEMA_ERROR", status: 500, description: "The page shape changed.", retryable: false },
	{ code: "NOT_FOUND", status: 404, description: "No item matches.", retryable: false },
] as const satisfies readonly OperationErrorCode[];

export const searchOperation = defineOperation<ProviderContext>()({
	descriptionKey: "operations.search.description",
	errorCodes: [...SESSION_ERROR_CODES, { code: "SHOP_DOWN", status: 502, description: "The shop is down.", retryable: false }],
	handler: async (input, ctx) => {
		if (input.query === "") throw new ProviderError("The page shape changed.", { code: "UPSTREAM_SCHEMA_ERROR" });
		if (input.query === "?") throw new ProviderError("No item matches.", { code: "NOT_FOUND" });
		return ctx.fetch(input.query);
	},
});
