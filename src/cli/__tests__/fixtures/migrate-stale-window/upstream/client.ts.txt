import type { ProviderContext } from "../index";

const TTL_MS = 60_000;
const STALE_IF_ERROR_MS = 6 * 60 * 60 * 1000;
export const CATALOG_STALE_MS = 24 * 60 * 60 * 1000;

export async function cachedJson(ctx: ProviderContext, path: string): Promise<unknown> {
	const result = await ctx.cache.getOrSet(ctx.cache.key("json", { path }), () => fetchJson(path), {
		ttlMs: TTL_MS,
		staleIfErrorMs: STALE_IF_ERROR_MS,
	});
	return result.value;
}

export async function cachedCatalog(ctx: ProviderContext): Promise<unknown> {
	const result = await ctx.cache.getOrSet("catalog", () => fetchJson("/catalog"), {
		ttlMs: TTL_MS,
		staleIfErrorMs: CATALOG_STALE_MS,
	});
	return result.value;
}

export async function orphanRead(ctx: ProviderContext): Promise<unknown> {
	return ctx.cache.getOrSet("orphan", () => fetchJson("/orphan"), { ttlMs: TTL_MS, staleIfErrorMs: 5 });
}
