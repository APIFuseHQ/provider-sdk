type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resource overrides previously embedded in the monorepo
// provider. A fixed replica count is omitted because the HPA owns scaling.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 2,
    targetCPUUtilizationPercentage: 42,
  },
  resources: {
    cpu: "500m",
    memory: "256Mi",
  },
} satisfies ProviderDeploymentConfig;

export default deployment;
