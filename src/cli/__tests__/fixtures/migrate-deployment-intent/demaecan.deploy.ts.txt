type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for demaecan
// (infra/k8s/generated/providers/demaecan/). `replicas` is intentionally
// absent: when an HPA is present the generator omits replicas from the
// Deployment spec. Proxy-vendor port ranges are derived from the provider's
// proxy policy, so they are not repeated as additional upstream TCP ports.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 2,
    targetCPUUtilizationPercentage: 42,
  },
  resources: {
    cpu: "500m",
    memory: "256Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
