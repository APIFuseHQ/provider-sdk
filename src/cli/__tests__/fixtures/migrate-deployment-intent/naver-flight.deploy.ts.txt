type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for naver-flight. `replicas` is
// intentionally absent because the generated HPA owns scaling and the
// Deployment spec omits replicas. Redis remains enabled because the resolved
// manifest injects the shared cache URL. Proxy egress uses centrally managed
// port ranges, so there are no provider-authored additional TCP ports.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 1,
    targetCPUUtilizationPercentage: 70,
  },
  resources: {
    cpu: "100m",
    memory: "1Gi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
