type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for danawa at the cutover base
// (infra/k8s/generated/providers/danawa/ at origin/main). `replicas` is
// intentionally absent because the generated workload is controlled by an
// HPA. Redis remains enabled because the resolved manifest injects its URL and
// egress rule. Proxy-vendor port ranges are derived from the provider proxy
// policy, so they are not authored as additional upstream TCP ports here.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 2,
    targetCPUUtilizationPercentage: 35,
  },
  resources: {
    cpu: "400m",
    memory: "256Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
