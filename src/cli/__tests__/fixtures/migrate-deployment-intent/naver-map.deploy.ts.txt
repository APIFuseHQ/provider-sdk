type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for naver-map. `replicas` is
// intentionally absent: when an HPA is present the generator omits replicas
// from the Deployment spec, so declaring it here would drift from what is
// applied. Redis remains enabled as required by the generated provider
// environment and proxy allocator. NodeMaven's gateway port ranges are
// derived from the preserved provider proxy policy, so they are not authored
// as network.additionalTcpPorts overrides.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 1,
    targetCPUUtilizationPercentage: 70,
  },
  resources: {
    cpu: "25m",
    memory: "128Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
