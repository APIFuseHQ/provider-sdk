type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for lh-notice
// (infra/k8s/generated/providers/lh-notice/). `replicas` is intentionally
// absent: when an HPA is present the generator omits replicas from the
// Deployment spec, so declaring it here would drift from what is applied.
// The FQDN policy carries only the standard 443/80 egress ports for
// apis.data.go.kr, so no network.additionalTcpPorts entry is needed. Redis
// caching is enabled because both operations serve through a shared
// 30-minute TTL cache with stale-if-error fallback.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 1,
    targetCPUUtilizationPercentage: 70,
  },
  resources: {
    cpu: "25m",
    memory: "128Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
