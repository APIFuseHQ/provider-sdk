type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// Mirrors the resolved manifest truth for rakuten-ichiba
// (infra/k8s/generated/providers/rakuten-ichiba/). `replicas` is intentionally
// absent: when an HPA is present the generator omits replicas from the
// Deployment spec, so declaring it here would drift from what is applied.
// The FQDN policy carries only the standard 443/80 egress ports, so no
// network.additionalTcpPorts entry is needed.
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 1,
    targetCPUUtilizationPercentage: 70,
  },
  resources: {
    cpu: "25m",
    memory: "128Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
