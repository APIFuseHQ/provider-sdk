// Provider deployment contract for the polyrepo reusable-deploy pipeline.
//
// Shape mirrors ProviderDeploymentConfigSchema in the monorepo-owned provider
// registry package. This split repo intentionally keeps the deploy contract as a
// plain object so the central deploy engine can validate it without making the
// private registry package a runtime dependency here.
//
// Values are authored against the RESOLVED manifests
// (infra/k8s/generated/providers/rakuten-travel/*.yaml), not against the
// monorepo-authored override keys, so regenerated manifests stay byte-identical.
//
// rakuten-travel is a plain REST provider over the Rakuten Web Service
// openapi.rakuten.co.jp endpoints (standard 443), with Redis-backed response
// caching (area classes / hotel chains are near-static documents).

type ProviderDeploymentConfig = {
  runtime: "shared" | "dedicated" | "browser";
  language: "typescript" | "python";
  replicas?: number;
  hpa?: {
    enabled: boolean;
    minReplicas?: number;
    maxReplicas?: number;
    targetCPUUtilizationPercentage?: number;
  };
  resources: {
    cpu: string;
    memory: string;
  };
  cache?: {
    redis?: {
      enabled: boolean;
      url?: string;
    };
  };
  network?: {
    additionalTcpPorts?: number[];
  };
};

// `replicas` is intentionally absent: when an HPA is present the generator
// omits replicas from the Deployment spec, so declaring it here would drift
// from what is actually applied (verified against
// infra/k8s/generated/providers/rakuten-travel/deployment.yaml).
const deployment = {
  runtime: "shared",
  language: "typescript",
  hpa: {
    enabled: true,
    minReplicas: 1,
    maxReplicas: 1,
    targetCPUUtilizationPercentage: 70,
  },
  resources: {
    cpu: "25m",
    memory: "128Mi",
  },
  cache: { redis: { enabled: true } },
} satisfies ProviderDeploymentConfig;

export default deployment;
