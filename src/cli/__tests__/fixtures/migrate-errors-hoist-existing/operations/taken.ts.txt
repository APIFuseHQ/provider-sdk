import { defineOperation } from "@apifuse/provider-sdk/provider";

import type { ProviderContext } from "../index";

const errors = ["local"];

export const takenOperation = defineOperation<ProviderContext>()({
	descriptionKey: "operations.taken.description",
	errorCodes: [{ code: "TAKEN", status: 400, description: "The file owns the name errors." }],
	handler: async (input, ctx) => ctx.fetch(errors[0] + input.id),
});
