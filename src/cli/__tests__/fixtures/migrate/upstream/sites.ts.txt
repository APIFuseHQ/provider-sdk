import { ProviderError, TransportError } from "@apifuse/provider-sdk/provider";

export function shopDown(detail: string): never {
	throw new ProviderError("The shop is down.", {
		code: "SHOP_DOWN",
		category: "upstream_http",
		retryable: false,
		details: { detail },
	});
}

export function shopDownAgain(): never {
	throw new ProviderError("The shop is still down.", { code: "SHOP_DOWN", category: "upstream_http" });
}

export function shopBusy(): never {
	throw new ProviderError("The shop is busy.", { code: "SHOP_BUSY", retryable: true });
}

export function shopBusyTransport(): never {
	throw new TransportError("The shop did not answer.", { code: "SHOP_BUSY" });
}

export function slotGoneInput(): never {
	throw new ProviderError("The slot is gone.", { code: "SLOT_GONE", category: "input_validation" });
}

export function slotGone(): never {
	throw new ProviderError("The slot is gone.", { code: "SLOT_GONE", category: "upstream_rejected" });
}

export function shopSlow(): never {
	throw new ProviderError("The shop timed out.", { code: "SHOP_SLOW", category: "timeout", retryable: false });
}

export function slotHeld(): never {
	throw new ProviderError("The slot is held.", { code: "SLOT_HELD", category: "timeout" });
}

export function slotHeldAgain(): never {
	throw new ProviderError("The slot is still held.", { code: "SLOT_HELD", category: "timeout" });
}

export function slotHeldBare(): never {
	throw new ProviderError("The slot is held by someone.", { code: "SLOT_HELD" });
}

export function upstreamDown(): never {
	throw new ProviderError("The shop answered 5xx.", { code: "UPSTREAM_ERROR" });
}

export function collect(): string[] {
	const errors: string[] = [];
	if (errors.length > 0) throw new ProviderError("The item was removed.", { code: "ITEM_GONE" });
	return errors;
}
