import { defineOperation } from "@apifuse/provider-sdk/provider";
import { z } from "zod";

import type { ProviderContext } from "../index";
import { SEARCH_ERRORS } from "./search";

const DetailInput = z.object({
	itemId: z.string().describe("The item id from a search result."),
});

export const detailOperation = defineOperation<ProviderContext>()({
	descriptionKey: "operations.detail.description",
	input: DetailInput,
	errorCodes: [
		...SEARCH_ERRORS,
		{ code: "ITEM_GONE", status: 410, description: "The item was removed.", retryable: false },
		{ code: "SHOP_DOWN", status: 502, description: "The shop is down.", retryable: false },
		{ code: "SHOP_BUSY", status: 503, description: "The shop is busy.", retryable: false },
		{ code: "SLOT_GONE", status: 410, description: "The slot is gone.", retryable: false },
		{ code: "SLOT_HELD", status: 409, description: "The slot is held.", retryable: false },
		{ code: "PAGE_CHANGED", status: 502, description: "The page changed.", retryable: true },
	],
	handler: async (input) => fetchItem(input.itemId),
});
