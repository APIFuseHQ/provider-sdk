import { defineOperation } from "@apifuse/provider-sdk/provider";

import type { ProviderContext } from "../index";

export const listOperation = defineOperation<ProviderContext>()({
	descriptionKey: "operations.list.description",
	errorCodes: [
		{ code: "SHOP_DOWN", status: 502, description: "The shop is down.", retryable: false },
		{ code: "NOT_FOUND", status: 404, description: "No item matches.", retryable: false },
		{ code: "BAD_ID", status: 400, description: "The id is malformed.", retryable: false },
	],
	handler: async (input, ctx) => ctx.fetch(input.id),
});
